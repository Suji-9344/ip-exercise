function validateForm()
{
let name=document.getElementById("name").value.trim();
let father=document.getElementById("father").value.trim();
let dob=document.getElementById("dob").value;
let gender=document.querySelector('input[name="gender"]:checked');
let phone=document.getElementById("phone").value.trim();
let email=document.getElementById("email").value.trim();
let address=document.getElementById("address").value.trim();
let qualification=document.getElementById("qualification").value;
let skills=document.getElementById("skills").value.trim();
let experience=document.getElementById("experience").value;
let username=document.getElementById("username").value.trim();
let password=document.getElementById("password").value;
let confirm=document.getElementById("confirm").value;
let resume=document.getElementById("resume").value;
document.querySelectorAll(".error").forEach(function(e){
e.innerHTML="";
});
let valid=true;
if(name==""){
document.getElementById("ename").innerHTML="Enter Full Name";
valid=false;
}
else if(!/^[A-Za-z ]+$/.test(name)){
document.getElementById("ename").innerHTML="Name must contain letters only";
valid=false;
}
if(father==""){
document.getElementById("efather").innerHTML="Enter Father Name";
valid=false;
}
else if(!/^[A-Za-z ]+$/.test(father)){
document.getElementById("efather").innerHTML="Father Name must contain letters only";
valid=false;
}
if(dob==""){
document.getElementById("edob").innerHTML="Select Date of Birth";
valid=false;
}
if(gender==null){
document.getElementById("egender").innerHTML="Select Gender";
valid=false;
}
if(phone==""){
document.getElementById("ephone").innerHTML="Enter Mobile Number";
valid=false;
}
else if(!/^[0-9]{10}$/.test(phone)){
document.getElementById("ephone").innerHTML="Mobile Number must contain 10 digits only";
valid=false;
}
if(email==""){
document.getElementById("eemail").innerHTML="Enter Email";
valid=false;
}
else if(!/^[^\s@]+@[^\s@]+\.com$/.test(email)){
document.getElementById("eemail").innerHTML="Email must be valid like name@example.com";
valid=false;
}
if(address==""){
document.getElementById("eaddress").innerHTML="Enter Address";
valid=false;
}

if(qualification==""){
document.getElementById("equalification").innerHTML="Select Qualification";
valid=false;
}
if(skills==""){
document.getElementById("eskills").innerHTML="Enter Skills";
valid=false;
}
if(experience==""){
document.getElementById("eexperience").innerHTML="Enter Experience";
valid=false;
}
else if(experience<0 || experience>50){
document.getElementById("eexperience").innerHTML="Experience must be between 0 and 50 years";
valid=false;
}
if(username==""){
document.getElementById("eusername").innerHTML="Enter Username";
valid=false;
}
if(password==""){
document.getElementById("epassword").innerHTML="Enter Password";
valid=false;
}
else if(password.length<6){
document.getElementById("epassword").innerHTML="Password must be at least 6 characters";
valid=false;
}
if(confirm==""){
document.getElementById("econfirm").innerHTML="Confirm Password";
valid=false;
}
else if(password!=confirm){
document.getElementById("econfirm").innerHTML="Password Not Matched";
valid=false;
}
if(resume==""){
document.getElementById("eresume").innerHTML="Upload Resume";
valid=false;
}
else if(!resume.match(/\.(pdf|doc|docx)$/i)){
document.getElementById("eresume").innerHTML="Only PDF/DOC/DOCX Allowed";
valid=false;
}
if(valid)
{
alert("Registration Successful");
return true;
}
else
{
return false;
}
}
